"""Research only: reflect opposed shading normals into each face hemisphere; no geometry repair claim."""
import bpy,json,sys,hashlib
import numpy as np
from pathlib import Path
request=json.loads(Path(sys.argv[-1]).read_text('utf-8'));output=Path(request['output']);output.mkdir(parents=True,exist_ok=False)
def sha(p):
 with Path(p).open('rb') as f:return hashlib.file_digest(f,'sha256').hexdigest()
source_hash=sha(request['source']);bpy.ops.wm.read_factory_settings(use_empty=True);bpy.ops.import_scene.gltf(filepath=request['source'])
objects=[o for o in bpy.context.scene.objects if o.type=='MESH'];changes=[]
for obj in objects:
 mesh=obj.data;mesh.update()
 normals=np.empty(len(mesh.loops)*3,dtype=np.float32);mesh.corner_normals.foreach_get('vector',normals);normals=normals.reshape(-1,3)
 original=normals.copy();faces=np.empty(len(mesh.polygons)*3,dtype=np.float32);mesh.polygons.foreach_get('normal',faces);faces=faces.reshape(-1,3)
 counts=np.empty(len(mesh.polygons),dtype=np.int32);mesh.polygons.foreach_get('loop_total',counts);face_normals=np.repeat(faces,counts,axis=0)
 dots=np.einsum('ij,ij->i',normals,face_normals);bad=dots< -1e-6
 normals[bad]-=2*dots[bad,None]*face_normals[bad]
 length=np.linalg.norm(normals,axis=1);normals=np.divide(normals,length[:,None],out=normals,where=length[:,None]>0)
 mesh.normals_split_custom_set(normals.tolist())
 changes.append(dict(name=obj.name,loops=len(normals),reflected_loops=int(bad.sum()),original_min_dot=float(dots.min()),finite=bool(np.isfinite(normals).all()),positions_faces_uv_edited=False))
bpy.ops.object.select_all(action='DESELECT')
for obj in objects:obj.select_set(True)
bpy.context.view_layer.objects.active=objects[0];bpy.ops.export_scene.gltf(filepath=str(output/'Derived_corner_normals.glb'),export_format='GLB',use_selection=True,export_animations=False,export_yup=True)
assert sha(request['source'])==source_hash
(output/'report.json').write_text(json.dumps(dict(status='RESEARCH_EXPORT_REQUIRES_VALIDATION',source_sha256=source_hash,changes=changes,scope='Only per-corner shading normals changed; no hole/topology/art repair or production integration claim.'),indent=2),'utf-8')
