def image_pixel_mutator(binding,channel):
    def change(d,b):
        image_index=d['textures'][d['materials'][0]['pbrMetallicRoughness'][binding]['index']]['source']
        _,layout=glb_layout(SOURCE);_,data=image_bytes(SOURCE,d,layout,d['images'][image_index])
        with Image.open(io.BytesIO(data)) as image:
            rgba=image.convert('RGBA');v=list(rgba.getpixel((0,0)));v[channel]=(v[channel]+1)%256;rgba.putpixel((0,0),tuple(v));out=io.BytesIO();rgba.save(out,format='PNG');encoded=out.getvalue()
        b.extend(b'\0'*((-len(b))%4));offset=len(b);b.extend(encoded)
        d['bufferViews'].append(dict(buffer=0,byteOffset=offset,byteLength=len(encoded)))
        d['images'][image_index]=dict(bufferView=len(d['bufferViews'])-1,mimeType='image/png')
    return change

def rewrite(source,target,mutate):
    doc,layout=glb_layout(source)
    with source.open('rb') as f:f.seek(layout[0]);binary=bytearray(f.read(doc['buffers'][0]['byteLength']))
    mutate(doc,binary)
    doc['buffers'][0]['byteLength']=len(binary)
    raw=json.dumps(doc,separators=(',',':')).encode();raw+=b' '*((-len(raw))%4);binary+=b'\0'*((-len(binary))%4)
    with target.open('wb') as f:
        f.write(struct.pack('<4sII',b'glTF',2,12+8+len(raw)+8+len(binary)))
        f.write(struct.pack('<II',len(raw),0x4e4f534a));f.write(raw)
        f.write(struct.pack('<II',len(binary),0x004e4942));f.write(binary)

def test_actual_factor_loss(tmp_path):
    target=tmp_path/'changed_factor.glb'
    rewrite(SOURCE,target,lambda d,b:d['materials'][0]['pbrMetallicRoughness'].update(roughnessFactor=.123456))
    r=p.compare(SOURCE,target);assert r['status']=='Fail' and r['checks']['core_factors'] is False
